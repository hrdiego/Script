(async () => {
    // ============================================================
    // CONFIGURACIÓN
    // ============================================================

    const ROOT = new URL(
	// cambiar el URL del SCORM
        "/courses/1/CIIN.1403P.226534.7138.P/content/_66082166_1/",
        location.origin
    ).href;
    
	// cambiar el nombre del archivo, siempre en .zip
    const ZIP_NAME = "SCORM_Tema_03_Tecnologia_Materiales.zip";

    console.log("📦 Iniciando recopilación del contenido...");
    console.log("📁 Ruta raíz:", ROOT);

    // ============================================================
    // CARGAR JSZIP
    // ============================================================

    async function cargarJSZip() {
        if (window.JSZip) return window.JSZip;

        await new Promise((resolve, reject) => {
            const script = document.createElement("script");

            script.src =
                "https://cdnjs.cloudflare.com/ajax/libs/jszip/3.10.1/jszip.min.js";

            script.onload = resolve;
            script.onerror = reject;

            document.head.appendChild(script);
        });

        if (!window.JSZip) {
            throw new Error("No se pudo cargar JSZip");
        }

        return window.JSZip;
    }

    const JSZip = await cargarJSZip();
    const zip = new JSZip();

    // ============================================================
    // CONTROL DE ARCHIVOS
    // ============================================================

    const visitados = new Set();
    const pendientes = [];
    const descargados = [];
    const errores = [];

    function perteneceAlSCORM(url) {
        return url.href.startsWith(ROOT);
    }

    function normalizarURL(url, base = ROOT) {
        try {
            const u = new URL(url, base);

            u.hash = "";

            return u;
        } catch {
            return null;
        }
    }

    function agregarPendiente(url, base = ROOT) {
        const u = normalizarURL(url, base);

        if (!u) return;

        if (!perteneceAlSCORM(u)) return;

        if (!visitados.has(u.href)) {
            pendientes.push(u.href);
        }
    }

    function rutaDentroDelZip(url) {
        const u = new URL(url);

        let ruta = u.pathname.replace(
            new URL(ROOT).pathname,
            ""
        );

        ruta = decodeURIComponent(ruta);

        if (!ruta || ruta.endsWith("/")) {
            ruta += "index.html";
        }

        return ruta;
    }

    // ============================================================
    // DETECTAR RECURSOS EN HTML
    // ============================================================

    function analizarHTML(texto, baseURL) {
        const parser = new DOMParser();

        const doc = parser.parseFromString(
            texto,
            "text/html"
        );

        // src
        doc.querySelectorAll(
            "[src]"
        ).forEach(el => {
            agregarPendiente(
                el.getAttribute("src"),
                baseURL
            );
        });

        // href
        doc.querySelectorAll(
            "a[href], link[href], area[href]"
        ).forEach(el => {
            const href = el.getAttribute("href");

            if (!href) return;

            if (
                href.startsWith("#") ||
                href.startsWith("mailto:") ||
                href.startsWith("tel:") ||
                href.startsWith("javascript:")
            ) return;

            agregarPendiente(href, baseURL);
        });

        // srcset
        doc.querySelectorAll(
            "[srcset]"
        ).forEach(el => {
            const srcset =
                el.getAttribute("srcset") || "";

            srcset
                .split(",")
                .map(x => x.trim().split(/\s+/)[0])
                .filter(Boolean)
                .forEach(url => {
                    agregarPendiente(
                        url,
                        baseURL
                    );
                });
        });

        // style=""
        doc.querySelectorAll(
            "[style]"
        ).forEach(el => {
            analizarCSS(
                el.getAttribute("style"),
                baseURL
            );
        });
    }

    // ============================================================
    // DETECTAR RECURSOS EN CSS
    // ============================================================

    function analizarCSS(texto, baseURL) {

        const regex =
            /url\(\s*['"]?([^'")]+)['"]?\s*\)/gi;

        let match;

        while (
            (match = regex.exec(texto)) !== null
        ) {
            agregarPendiente(
                match[1],
                baseURL
            );
        }

        const imports =
            /@import\s+(?:url\()?['"]?([^'"\)\s]+)['"]?\)?/gi;

        while (
            (match = imports.exec(texto)) !== null
        ) {
            agregarPendiente(
                match[1],
                baseURL
            );
        }
    }

    // ============================================================
    // ANALIZAR MANIFEST SCORM
    // ============================================================

    function analizarManifest(texto, baseURL) {

        const parser = new DOMParser();

        const xml = parser.parseFromString(
            texto,
            "application/xml"
        );

        xml.querySelectorAll(
            "[href]"
        ).forEach(el => {
            agregarPendiente(
                el.getAttribute("href"),
                baseURL
            );
        });
    }

    // ============================================================
    // ARCHIVOS PRINCIPALES CONOCIDOS
    // ============================================================

    [
        "imsmanifest.xml",

        "index.html",

        "scorm-api-wrapper.js",
        "scorm-launcher.js",
        "scorm-bridge.js",

        "design-tokens.css",
        "contenido.css",
        "componentes.css",
        "estructura-mit.css",
        "medios.css",

        "figuras.js",
        "estructura.js",
        "medios.js",

        "logo-upn.svg"
    ].forEach(archivo => {
        agregarPendiente(archivo);
    });

    // También añadimos las páginas que ya sabemos que existen
    [
        "tema-3/index.html",
        "tema-3/3-1.html",
        "tema-3/3-2.html",
        "tema-3/3-3.html",
        "tema-3/3-4.html"
    ].forEach(archivo => {
        agregarPendiente(archivo);
    });

    // ============================================================
    // DESCARGA RECURSIVA
    // ============================================================

    while (pendientes.length > 0) {

        const url = pendientes.shift();

        if (visitados.has(url)) continue;

        visitados.add(url);

        const rutaZip =
            rutaDentroDelZip(url);

        console.log(
            `⬇️ [${visitados.size}] ${rutaZip}`
        );

        try {

            const respuesta = await fetch(
                url,
                {
                    credentials: "include"
                }
            );

            if (!respuesta.ok) {
                console.warn(
                    `⚠️ ${respuesta.status}: ${rutaZip}`
                );

                errores.push({
                    url,
                    status: respuesta.status
                });

                continue;
            }

            const tipo =
                respuesta.headers.get("content-type") || "";

            const buffer =
                await respuesta.arrayBuffer();

            zip.file(
                rutaZip,
                buffer
            );

            descargados.push(rutaZip);

            // Analizar archivos de texto
            if (
                tipo.includes("text/html") ||
                /\.html?$/i.test(rutaZip)
            ) {

                const texto =
                    new TextDecoder().decode(buffer);

                analizarHTML(
                    texto,
                    url
                );
            }

            else if (
                tipo.includes("text/css") ||
                /\.css$/i.test(rutaZip)
            ) {

                const texto =
                    new TextDecoder().decode(buffer);

                analizarCSS(
                    texto,
                    url
                );
            }

            else if (
                /imsmanifest\.xml$/i.test(rutaZip)
            ) {

                const texto =
                    new TextDecoder().decode(buffer);

                analizarManifest(
                    texto,
                    url
                );
            }

        } catch (error) {

            console.error(
                "❌ Error:",
                rutaZip,
                error
            );

            errores.push({
                url,
                error: String(error)
            });
        }
    }

    // ============================================================
    // INFORME
    // ============================================================

    const informe = {
        fecha: new Date().toISOString(),
        raiz: ROOT,
        archivos_descargados: descargados,
        archivos_con_error: errores
    };

    zip.file(
        "_informe_descarga.json",
        JSON.stringify(
            informe,
            null,
            2
        )
    );

    console.log("");
    console.log("========================================");
    console.log("📦 CREANDO ARCHIVO ZIP...");
    console.log(`✅ Archivos: ${descargados.length}`);
    console.log(`⚠️ Errores: ${errores.length}`);
    console.log("========================================");

    // ============================================================
    // CREAR ZIP
    // ============================================================

    const contenidoZip =
        await zip.generateAsync({
            type: "blob",
            compression: "DEFLATE",
            compressionOptions: {
                level: 6
            }
        });

    const blobURL =
        URL.createObjectURL(contenidoZip);

    const enlace =
        document.createElement("a");

    enlace.href = blobURL;
    enlace.download = ZIP_NAME;

    document.body.appendChild(enlace);
    enlace.click();
    enlace.remove();

    setTimeout(() => {
        URL.revokeObjectURL(blobURL);
    }, 10000);

    console.log("");
    console.log("🎉 DESCARGA TERMINADA");
    console.log(`📦 Archivo: ${ZIP_NAME}`);
    console.log(`📁 Total incluidos: ${descargados.length}`);
})();
